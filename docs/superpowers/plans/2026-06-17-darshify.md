```ts
export type Repeat = "off" | "all" | "one";
export const nextIndex = (i: number, len: number, repeat: Repeat): number => {
  if (len === 0) return -1;
  if (repeat === "one") return i;
  if (i + 1 < len) return i + 1;
  return repeat === "all" ? 0 : -1;
};
export const prevIndex = (i: number, _len: number): number =>
  Math.max(0, i - 1);
// deterministic seeded shuffle (mulberry32)
export const shuffleOrder = (len: number, seed: number): number[] => {
  const a = Array.from({ length: len }, (_, i) => i);
  let s = seed >>> 0;
  const rand = () => {
    s |= 0;
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
  for (let i = len - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
};
```

```tsx
import {
  createContext,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
  type ReactNode,
} from "react";
import { tracks, trackById, type Track } from "../data/library";
import { nextIndex, prevIndex, shuffleOrder, type Repeat } from "./engine";

interface PlayerState {
  current?: Track;
  isPlaying: boolean;
  progress: number;
  volume: number;
  repeat: Repeat;
  shuffle: boolean;
  queue: string[];
  likes: Set<string>;
  play: (track: Track, context?: string[]) => void;
  toggle: () => void;
  next: () => void;
  prev: () => void;
  seek: (sec: number) => void;
  setVolume: (v: number) => void;
  cycleRepeat: () => void;
  toggleShuffle: () => void;
  enqueue: (id: string) => void;
  toggleLike: (id: string) => void;
}
const Ctx = createContext<PlayerState | null>(null);
export const usePlayer = () => {
  const c = useContext(Ctx);
  if (!c) throw new Error("usePlayer outside provider");
  return c;
};

export function PlayerProvider({ children }: { children: ReactNode }) {
  const [order, setOrder] = useState<string[]>(tracks.map((t) => t.id)); // active queue (ids, possibly shuffled)
  const [pos, setPos] = useState(0);
  const [isPlaying, setPlaying] = useState(false);
  const [progress, setProgress] = useState(0);
  const [volume, setVol] = useState(0.8);
  const [repeat, setRepeat] = useState<Repeat>("off");
  const [shuffle, setShuffle] = useState(false);
  const [likes, setLikes] = useState<Set<string>>(new Set());
  const raf = useRef(0);

  const current = trackById(order[pos]);

  const play = (track: Track, context: string[] = tracks.map((t) => t.id)) => {
    let q = context;
    if (shuffle) {
      const ord = shuffleOrder(context.length, 7);
      q = ord.map((i) => context[i]);
    }
    const idx = q.indexOf(track.id);
    setOrder(q);
    setPos(idx < 0 ? 0 : idx);
    setProgress(0);
    setPlaying(true);
  };
  const toggle = () => setPlaying((p) => !p);
  const next = () => {
    const i = nextIndex(pos, order.length, repeat);
    if (i === -1) {
      setPlaying(false);
      return;
    }
    setPos(i);
    setProgress(0);
  };
  const prev = () => {
    if (progress > 3) {
      setProgress(0);
      return;
    }
    setPos((p) => prevIndex(p, order.length));
    setProgress(0);
  };
  const seek = (sec: number) => setProgress(sec);
  const cycleRepeat = () =>
    setRepeat((r) => (r === "off" ? "all" : r === "all" ? "one" : "off"));
  const toggleShuffle = () => setShuffle((s) => !s);
  const enqueue = (id: string) => setOrder((o) => [...o, id]);
  const toggleLike = (id: string) =>
    setLikes((s) => {
      const n = new Set(s);
      n.has(id) ? n.delete(id) : n.add(id);
      return n;
    });

  // progress ticker
  useEffect(() => {
    if (!isPlaying || !current) return;
    let last = performance.now();
    const tick = (t: number) => {
      const dt = (t - last) / 1000;
      last = t;
      setProgress((p) => {
        const np = p + dt;
        if (np >= current.durationSec) {
          const i = nextIndex(pos, order.length, repeat);
          if (i === -1) {
            setPlaying(false);
            return current.durationSec;
          }
          setPos(i);
          return 0;
        }
        return np;
      });
      raf.current = requestAnimationFrame(tick);
    };
    raf.current = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf.current);
  }, [isPlaying, current, pos, order.length, repeat]);

  const value = useMemo<PlayerState>(
    () => ({
      current,
      isPlaying,
      progress,
      volume,
      repeat,
      shuffle,
      queue: order,
      likes,
      play,
      toggle,
      next,
      prev,
      seek,
      setVolume: setVol,
      cycleRepeat,
      toggleShuffle,
      enqueue,
      toggleLike,
    }),
    [current, isPlaying, progress, volume, repeat, shuffle, order, likes],
  );

  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}
```
